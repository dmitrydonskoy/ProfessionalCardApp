#include <iostream>
#include <map>
#include <sstream>
using namespace std;

int main() {
    string text;
    cout << "Введите строку: ";
    getline(cin, text);

    map<string, int> freq;
    stringstream ss(text);
    string word;

    while (ss >> word) {
        freq[word]++;
    }

    cout << "Частота слов:\n";
    for (const auto& pair : freq) {
        cout << pair.first << ": " << pair.second << endl;
    }

    return 0;
}

#include <iostream>
#include <set>
using namespace std;

int main() {
    int n;
    cout << "Сколько чисел вы хотите ввести? ";
    cin >> n;

    set<int> uniqueNumbers;
    int num;

    cout << "Введите числа:\n";
    for (int i = 0; i < n; ++i) {
        cin >> num;
        uniqueNumbers.insert(num); // дубликаты игнорируются автоматически
    }

    cout << "Количество уникальных чисел: " << uniqueNumbers.size() << endl;

    return 0;
}

#include <iostream>
#include <map>
#include <vector>
#include <string>
#include <numeric> // для accumulate
using namespace std;

int main() {
    map<string, vector<int>> grades;
    int choice;

    do {
        cout << "\n1. Добавить оценку студенту\n";
        cout << "2. Посмотреть все оценки\n";
        cout << "3. Посмотреть среднюю оценку\n";
        cout << "0. Выход\n";
        cout << "Выбор: ";
        cin >> choice;
        cin.ignore();

        string name;
        int grade;

        switch (choice) {
            case 1:
                cout << "Имя студента: ";
                getline(cin, name);
                cout << "Оценка: ";
                cin >> grade;
                grades[name].push_back(grade);
                break;

            case 2:
                for (const auto& [key, vec] : grades) {
                    cout << key << ": ";
                    for (int g : vec) cout << g << " ";
                    cout << endl;
                }
                break;

            case 3:
                cout << "Имя студента: ";
                getline(cin, name);
                if (grades.count(name)) {
                    vector<int>& v = grades[name];
                    double avg = accumulate(v.begin(), v.end(), 0.0) / v.size();
                    cout << "Средняя оценка: " << avg << endl;
                } else {
                    cout << "Нет данных о студенте.\n";
                }
                break;

            case 0: break;
            default: cout << "Неверный выбор.\n";
        }
    } while (choice != 0);

    return 0;
}
